import { useState } from 'react';
import { motion } from 'framer-motion';
import { useQuery, useMutation } from '@tanstack/react-query';
import { Sparkles, FileText, Loader2, Upload, Save } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';
import { Textarea } from '@/components/ui/textarea';
import { useStore } from '@/store/useStore';
import { toast } from 'sonner';
import api from '@/utils/api';

export const NoteSummarizer = () => {
  const [noteText, setNoteText] = useState('');
  const [file, setFile] = useState(null);
  const { theme, user } = useStore();

  // Mutation for summarization
  const summarizeMutation = useMutation({
    mutationFn: async (text) => {
      const res = await api.post('/api/notes/summarize', { text });
      return res.data;
    },
    onSuccess: (data) => {
      toast.success('Notes summarized successfully!');
    },
    onError: (error) => {
      toast.error('Failed to summarize notes');
      console.error(error);
    },
  });

  // File upload handler
  const handleFileUpload = async (e) => {
    const uploadedFile = e.target.files[0];
    if (!uploadedFile) return;

    setFile(uploadedFile);
    const reader = new FileReader();
    
    reader.onload = (event) => {
      setNoteText(event.target.result);
    };
    
    reader.readAsText(uploadedFile);
  };

  // Summarize handler
  const handleSummarize = () => {
    if (!noteText.trim()) {
      toast.error('Please enter some notes to summarize');
      return;
    }
    summarizeMutation.mutate(noteText);
  };

  return (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ duration: 0.5 }}
      className="relative z-10 max-w-4xl mx-auto p-6"
    >
      <Card className="backdrop-blur-lg bg-white/90 dark:bg-slate-900/90 
                       border-2 border-indigo-200 dark:border-indigo-800 
                       shadow-2xl">
        <div className="p-8">
          {/* Header */}
          <div className="flex items-center gap-3 mb-6">
            <Sparkles className="w-8 h-8 text-indigo-600 dark:text-indigo-400" />
            <h2 className="text-3xl font-bold bg-gradient-to-r 
                           from-indigo-600 to-purple-600 
                           dark:from-indigo-400 dark:to-purple-400 
                           bg-clip-text text-transparent">
              Smart Note Summarizer
            </h2>
          </div>

          {/* File Upload */}
          <div className="mb-4">
            <label className="flex items-center justify-center gap-2 p-4 
                              border-2 border-dashed border-slate-300 
                              dark:border-slate-600 rounded-lg cursor-pointer 
                              hover:border-indigo-500 dark:hover:border-indigo-400 
                              transition-colors">
              <Upload className="w-5 h-5" />
              <span>Upload Text/PDF File</span>
              <input
                type="file"
                accept=".txt,.pdf"
                onChange={handleFileUpload}
                className="hidden"
              />
            </label>
            {file && (
              <p className="mt-2 text-sm text-slate-600 dark:text-slate-400">
                Selected: {file.name}
              </p>
            )}
          </div>

          {/* Text Input */}
          <Textarea
            value={noteText}
            onChange={(e) => setNoteText(e.target.value)}
            className="w-full h-64 p-4 rounded-lg border-2 
                       border-slate-200 dark:border-slate-700 
                       bg-white dark:bg-slate-800 
                       focus:border-indigo-500 dark:focus:border-indigo-400 
                       transition-colors resize-none"
            placeholder="Or paste your notes here..."
          />

          {/* Character Count */}
          <div className="mt-2 text-right text-sm text-slate-500">
            {noteText.length} characters
          </div>

          {/* Summarize Button */}
          <Button
            onClick={handleSummarize}
            disabled={!noteText || summarizeMutation.isPending}
            className="mt-6 w-full bg-gradient-to-r from-indigo-600 to-purple-600 
                       hover:from-indigo-700 hover:to-purple-700 
                       text-white font-semibold py-3 text-lg
                       disabled:opacity-50 disabled:cursor-not-allowed"
          >
            {summarizeMutation.isPending ? (
              <>
                <Loader2 className="w-5 h-5 mr-2 animate-spin" />
                Summarizing with AI...
              </>
            ) : (
              <>
                <Sparkles className="w-5 h-5 mr-2" />
                Summarize Notes
              </>
            )}
          </Button>

          {/* Summary Result */}
          {summarizeMutation.data && (
            <motion.div
              initial={{ opacity: 0, height: 0 }}
              animate={{ opacity: 1, height: 'auto' }}
              transition={{ duration: 0.5 }}
              className="mt-8 p-6 rounded-lg bg-gradient-to-br 
                         from-indigo-50 to-purple-50 
                         dark:from-indigo-950/30 dark:to-purple-950/30 
                         border-l-4 border-indigo-600"
            >
              <div className="flex items-center justify-between mb-4">
                <h3 className="text-xl font-semibold text-indigo-900 
                               dark:text-indigo-100">
                  AI Summary
                </h3>
                <Button variant="outline" size="sm">
                  <Save className="w-4 h-4 mr-2" />
                  Save Summary
                </Button>
              </div>
              
              <div className="prose dark:prose-invert max-w-none">
                <p className="text-slate-700 dark:text-slate-300 leading-relaxed">
                  {summarizeMutation.data.summary}
                </p>
              </div>

              {/* Key Points */}
              {summarizeMutation.data.keyPoints && (
                <div className="mt-4">
                  <h4 className="font-semibold mb-2 text-indigo-900 
                                 dark:text-indigo-100">
                    Key Points:
                  </h4>
                  <ul className="list-disc list-inside space-y-1">
                    {summarizeMutation.data.keyPoints.map((point, idx) => (
                      <li key={idx} className="text-slate-700 dark:text-slate-300">
                        {point}
                      </li>
                    ))}
                  </ul>
                </div>
              )}
            </motion.div>
          )}
        </div>
      </Card>
    </motion.div>
  );
};


THREE.JS BACKGROUND COMPONENT (Dark Mode)

File: client/src/components/3d/DarkBackground.jsx

import { useRef, useMemo } from 'react';
import { Canvas, useFrame } from '@react-three/fiber';
import { Points, PointMaterial } from '@react-three/drei';
import * as random from 'maath/random';

function StarField() {
  const ref = useRef();
  
  // Generate random sphere of stars
  const sphere = useMemo(() => {
    const positions = random.inSphere(new Float32Array(5000 * 3), { 
      radius: 1.5 
    });
    return positions;
  }, []);

  // Rotate stars slowly
  useFrame((state, delta) => {
    ref.current.rotation.x -= delta / 10;
    ref.current.rotation.y -= delta / 15;
  });

  return (
    <group rotation={[0, 0, Math.PI / 4]}>
      <Points 
        ref={ref} 
        positions={sphere} 
        stride={3} 
        frustumCulled={false}
      >
        <PointMaterial
          transparent
          color="#818cf8"
          size={0.002}
          sizeAttenuation={true}
          depthWrite={false}
          blending={2} // Additive blending
        />
      </Points>
    </group>
  );
}

function ConnectedNodes() {
  const ref = useRef();
  
  const particles = useMemo(() => {
    const positions = [];
    for (let i = 0; i < 100; i++) {
      positions.push(
        (Math.random() - 0.5) * 3,
        (Math.random() - 0.5) * 3,
        (Math.random() - 0.5) * 3
      );
    }
    return new Float32Array(positions);
  }, []);

  useFrame((state) => {
    ref.current.rotation.y = state.clock.elapsedTime * 0.05;
  });

  return (
    <Points ref={ref} positions={particles} stride={3}>
      <PointMaterial
        transparent
        color="#a78bfa"
        size={0.015}
        sizeAttenuation={true}
        depthWrite={false}
      />
    </Points>
  );
}

export const DarkBackground = () => {
  return (
    <div className="fixed inset-0 -z-10">
      <Canvas camera={{ position: [0, 0, 1] }}>
        <StarField />
        <ConnectedNodes />
      </Canvas>
      
      {/* Gradient Overlay */}
      <div className="absolute inset-0 bg-gradient-to-br 
                      from-slate-900 via-purple-900/20 to-slate-900 
                      pointer-events-none" />
    </div>
  );
};

import { Points, PointMaterial } from '@react-three/drei';

function FloatingParticles() {
  const ref = useRef();
  
  const particles = useMemo(() => {
    const positions = [];
    for (let i = 0; i < 200; i++) {
      positions.push(
        (Math.random() - 0.5) * 10,
        Math.random() * 10 - 5,
        (Math.random() - 0.5) * 10
      );
    }
    return new Float32Array(positions);
  }, []);

  useFrame((state, delta) => {
    ref.current.rotation.y += delta * 0.05;
    
    // Floating animation
    const positions = ref.current.geometry.attributes.position.array;
    for (let i = 1; i < positions.length; i += 3) {
      positions[i] += Math.sin(state.clock.elapsedTime + i) * 0.001;
    }
    ref.current.geometry.attributes.position.needsUpdate = true;
  });

  return (
    <Points ref={ref} positions={particles} stride={3}>
      <PointMaterial
        transparent
        color="#6366f1"
        size={0.025}
        sizeAttenuation={true}
        depthWrite={false}
        opacity={0.6}
      />
    </Points>
  );
}

export const LightBackground = () => {
  return (
    <div className="fixed inset-0 -z-10">
      <Canvas camera={{ position: [0, 0, 5] }}>
        <ambientLight intensity={0.5} />

      </Canvas>

      <div className="absolute inset-0 bg-gradient-to-br 
                      from-blue-50 via-purple-50 to-pink-50 
                      pointer-events-none" />
    </div>
  );
};

import { GoogleGenerativeAI } from '@google/generative-ai';
import Note from '../models/Note.js';
import User from '../models/User.js';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);

export const summarizeNote = async (req, res) => {
  try {
    const { text } = req.body;
    const userId = req.user.id;

    // Validate input
    if (!text || text.trim().length < 50) {
      return res.status(400).json({
        success: false,
        message: 'Please provide at least 50 characters of text',
      });
    }

    // Initialize Gemini model
    const model = genAI.getGenerativeModel({ model: 'gemini-pro' });
    
    // Create detailed prompt
    const prompt = `
      You are an expert study assistant. Analyze and summarize the following 
      student notes. Your summary should:
      
      1. Extract the main concepts and key ideas
      2. Organize information hierarchically
      3. Highlight important definitions, formulas, or facts
      4. Use clear, concise language
      5. Format with bullet points for readability
      
      Notes to summarize:
      ${text}
      
      Provide your response in the following JSON format:
      {
        "summary": "A comprehensive paragraph summary",
        "keyPoints": ["Point 1", "Point 2", "Point 3"],
        "difficulty": "beginner|intermediate|advanced",
        "estimatedReadTime": "X minutes"
      }
    `;

    // Generate content
    const result = await model.generateContent(prompt);
    const responseText = result.response.text();
    
    // Parse JSON response
    const cleanedResponse = responseText
      .replace(/```json\n?/g, '')
      .replace(/```\n?/g, '')
      .trim();
    
    const aiResponse = JSON.parse(cleanedResponse);

    // Save to database
    const note = await Note.create({
      user: userId,
      originalText: text,
      summary: aiResponse.summary,
      keyPoints: aiResponse.keyPoints,
      difficulty: aiResponse.difficulty,
      estimatedReadTime: aiResponse.estimatedReadTime,
      createdAt: new Date(),
    });

    // Update user's note count
    await User.findByIdAndUpdate(userId, {
      $inc: { totalNotes: 1 },
    });

    res.json({
      success: true,
      ...aiResponse,
      noteId: note._id,
    });
  } catch (error) {
    console.error('Summarization error:', error);
    res.status(500).json({
      success: false,
      message: 'Failed to summarize notes. Please try again.',
      error: error.message,
    });
  }
};

export const generateQuiz = async (req, res) => {
  try {
    const { noteId, difficulty = 'medium', questionCount = 5 } = req.body;
    const userId = req.user.id;
    
    // Find the note
    const note = await Note.findOne({ _id: noteId, user: userId });
    if (!note) {
      return res.status(404).json({ 
        success: false, 
        message: 'Note not found' 
      });
    }

    // Initialize Gemini
    const model = genAI.getGenerativeModel({ model: 'gemini-pro' });
    
    // Create quiz generation prompt
    const prompt = `
      Generate ${questionCount} multiple-choice questions based on these study notes.
      Difficulty level: ${difficulty}
      
      Requirements:
      - Questions should test understanding, not just memorization
      - 4 options per question (A, B, C, D)
      - Only one correct answer
      - Include brief explanations for correct answers
      - Mix question types (definition, application, analysis)
      
      Notes:
      ${note.originalText}
      
      Return ONLY valid JSON in this exact format:
      [
        {
          "question": "What is...?",
          "options": ["Option A", "Option B", "Option C", "Option D"],
          "correctAnswer": 0,
          "explanation": "Option A is correct because...",
          "difficulty": "easy|medium|hard"
        }
      ]
    `;

    const result = await model.generateContent(prompt);
    let quizData = result.response.text();
    
    // Clean and parse JSON
    quizData = quizData
      .replace(/```json\n?/g, '')
      .replace(/```\n?/g, '')
      .trim();
    
    const questions = JSON.parse(quizData);

    // Save quiz to database
    const Quiz = (await import('../models/Quiz.js')).default;
    const quiz = await Quiz.create({
      user: userId,
      note: noteId,
      title: `Quiz: ${note.title || 'Untitled'}`,
      questions,
      difficulty,
      createdAt: new Date(),
    });

    res.json({
      success: true,
      quiz: {
        id: quiz._id,
        questions,
        totalQuestions: questions.length,
      },
    });
  } catch (error) {
    console.error('Quiz generation error:', error);
    res.status(500).json({ 
      success: false,
      message: 'Failed to generate quiz',
      error: error.message,
    });
  }
};

export const getNoteHistory = async (req, res) => {
  try {
    const userId = req.user.id;
    const { page = 1, limit = 10 } = req.query;

    const notes = await Note.find({ user: userId })
      .sort({ createdAt: -1 })
      .limit(limit * 1)
      .skip((page - 1) * limit)
      .select('-originalText'); // Don't send full text in list

    const total = await Note.countDocuments({ user: userId });

    res.json({
      success: true,
      notes,
      pagination: {
        currentPage: page,
        totalPages: Math.ceil(total / limit),
        totalNotes: total,
      },
    });
  } catch (error) {
    console.error('Fetch notes error:', error);
    res.status(500).json({
      success: false,
      message: 'Failed to fetch notes',
    });
  }
};
